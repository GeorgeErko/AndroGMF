   GetProperties(Objects[Objects.Count-1]);

procedure TPropEditorForm.ObjInsp1EditButtonClick(Sender: TObject);
var OIT:TOIType;S:String;St:TStrings;I:Integer;
begin
 With TObjInsp2(Sender) do begin
  OIT:=NameType[Cells[0,Row]];
  S:=Cells[1,Row];
  If OIT.FPropEditor<>nil then begin
   If OIT.FPropEditor is TTreeEditor then begin
    S:=Cells[0,Row];
    St:=TStringList.Create;
    OIT.FPropEditor.Execute(S,St);
    For I:=0 to St.Count-1 do If Pos('*',St[I])=0 then St[I]:='##'+St[I];

    exit;
   end else

  end;
  If S='' then Cells[1,Row]:=byLayer else Cells[1,Row]:=S;
  SetProperty(Cells[0,Row],Cells[1,Row]);
 end;
end;

    SetProperty(propName,propValue);

   Windows.SetFocus(TForm2(GTwgForm).HWndParent);
  end;
 end;
end;

procedure TPropEditorForm.SetProperties(propNames, propValues: String);
var I,J,propSettingCount,Index:Integer;Undo:TUndo;Names,Values:TStrings;
begin
 Undo:=TwgForm.Undo;
 Undo.StartTransAction;
 Undo.AddUndoItem(TPrimUndo.Create(GTwgForm,LU_ModifiedPrim,'SetProperty...'+PropNames+'='+PropValues));
 Names:=TStringList.Create;Values:=TStringList.Create;
 Names.Text:=propNames;Values.Text:= propValues;
 try
  propSettingCount:=0;                                                               
//  Writeln(propName,' = ',propValue);
  If Objects<>nil then                           
   For I:=0 to Objects.Count-1 do begin                                      
     TPrimUndo(Undo.Last).AddModifiedPrim(Objects[I]);
    For J:=0 to Names.Count-1 do begin
     If Values.Count-1<J then begin If TTD(Objects[I]).SetProperty(Names[J],'') then Inc(propSettingCount) end else
     If TTD(Objects[I]).SetProperty(Names[J],Values[J]) then Inc(propSettingCount);
    end;
   end;
   For J:=0 to Names.Count-1 do begin
    Index:=ObjInsp[0].FindRowByName(Names[J]);

   propSettingCount:=0;

      TPrimUndo(Undo.Last).AddModifiedPrim(Objects[I]);

procedure TPropEditorForm.sbDownClick(Sender: TObject);
var Props:TProperties;I,Index:Integer;propName:String;
    P:PCollection;
begin
 propName:=ObjInsp[0].Cells[0,ObjInsp[0].Row];
 For I:=0 to Objects.Count-1 do begin
  If TObject(Objects[I]) is TLot then Props:=TLot(Objects[I]).Properties else Props:=TPointDot(Objects[I]).Properties;
  If TComponent(Sender).Tag = 1 then Props.Up(propName) else
  If TComponent(Sender).Tag = 2 then Props.Down(propName) else begin
   If MessageConfirm('Удалить свойство "'+propName+'" ?') = mrNo then exit;
   Props.DeleteProperty(propName);
  end;
 end;
 Index:=ObjInsp[0].Row;                          
 P:=PCollection.Create(1);
  For I:=0 to Objects.Count-1 do P.Insert(Objects[I]);
  Objects.Locked:=True;
  Objects.DeleteAll;
  For I:=0 to P.Count-1 do begin
   Objects.Insert(P[I]);
  end;
  Objects.Locked:=False;
  Objects.Update;
 P.DeleteAll;P.Free;
 If TComponent(Sender).Tag = 0 then If Index<ObjInsp[0].RowCount then ObjInsp[0].Row:=Index else ObjInsp[0].Row:=ObjInsp[0].RowCount-1
 else
 For I:=0 to ObjInsp[0].RowCount-1 do If ObjInsp[0].Cells[0,I]=propName then ObjInsp[0].Row:=I;
end;

   For I:=0 to P.Count-1 do begin

   Objects.Update;
  P.DeleteAll;P.Free;
  If Index<ObjInsp[0].RowCount then ObjInsp[0].Row:=Index else ObjInsp[0].Row:=ObjInsp[0].RowCount-1
 end;
end;                           

procedure TPropEditorForm.ObjInsp1MouseMove(Sender: TObject;
  Shift: TShiftState; X, Y: Integer);
begin
 If xMovePoint <> 0 then begin
  If X < 25 then exit;
  If objInsp1.Width - X < 25 then exit;
  objInsp1.ColWidths[0] := X; xMovePoint := X;
 end;
// ищем разделитель столбцов
 If abs(X - ObjInsp1.ColWidths[0]) < 3 then
  ObjInsp1.Cursor := crHSplit else ObjInsp1.Cursor := crDefault;
end;

procedure TPropEditorForm.ObjInsp1MouseUp(Sender: TObject;
  Button: TMouseButton; Shift: TShiftState; X, Y: Integer);
begin
 If xMovePoint <> 0 then begin
  xMovePoint := 0;
  GWriteInteger(Name + '_ColWidth', ObjInsp1.ColWidths[0]);
 end;
end;
